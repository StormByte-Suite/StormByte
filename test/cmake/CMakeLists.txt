cmake_minimum_required(VERSION 3.28)
project(StormByteDiscoveryTests NONE)

if(WIN32 AND NOT STORMBYTE_TEST_MINGW)
	set(_library_prefix "")
	set(_suffix ".lib")
else()
	set(_library_prefix "lib")
	if(STORMBYTE_TEST_STATIC)
		set(_suffix ".a")
	elseif(WIN32)
		set(_suffix ".dll.a")
	else()
		set(_suffix ".so")
	endif()
endif()
set(CMAKE_FIND_LIBRARY_PREFIXES "${_library_prefix}")
set(CMAKE_FIND_LIBRARY_SUFFIXES "${_suffix}")
set(_prefix "${CMAKE_CURRENT_BINARY_DIR}/prefix")
set(_modules "${_prefix}/share/cmake/Modules")
file(MAKE_DIRECTORY "${_modules}" "${_prefix}/lib" "${_prefix}/bin" "${_prefix}/include/StormByte")

function(write_fixture_library path shared)
	if(WIN32 AND NOT STORMBYTE_TEST_MINGW)
		if(shared)
			get_filename_component(_name "${path}" NAME_WLE)
			if(ARGC GREATER 2)
				set(_name "${ARGV2}")
			endif()
			set(_symbols "__IMPORT_DESCRIPTOR_${_name}")
		else()
			if(ARGC GREATER 2)
				set(_dependency "${ARGV2}")
			else()
				set(_dependency "UnrelatedDependency")
			endif()
			set(_symbols "??_C@__IMPORT_DESCRIPTOR_decoy" "__IMPORT_DESCRIPTOR_${_dependency}")
		endif()
		execute_process(
			COMMAND "${STORMBYTE_COFF_FIXTURE_GENERATOR}" "${path}" ${_symbols}
			RESULT_VARIABLE _result
			ERROR_VARIABLE _error
		)
		if(NOT _result STREQUAL "0")
			message(FATAL_ERROR "Could not create fixture ${path}: ${_result}\n${_error}")
		endif()
	else()
		file(WRITE "${path}" "")
	endif()
endfunction()

function(check_visibility target definition shared)
	get_target_property(_definitions "${target}" INTERFACE_COMPILE_DEFINITIONS)
	if(shared)
		if(NOT _definitions OR NOT definition IN_LIST _definitions)
			message(FATAL_ERROR "Shared target ${target} needs ${definition}: ${_definitions}")
		endif()
	elseif(_definitions MATCHES "STORMBYTE.*_SHARED")
		message(FATAL_ERROR "Static target ${target} must not define shared visibility: ${_definitions}")
	endif()
endfunction()

if(STORMBYTE_TEST_STATIC)
	set(_shared FALSE)
else()
	set(_shared TRUE)
endif()

file(COPY
	"${STORMBYTE_CMAKE_DIR}/FindStormByte.cmake"
	"${STORMBYTE_CMAKE_DIR}/StormByteCommon.cmake"
	DESTINATION "${_modules}"
)
file(WRITE "${_prefix}/include/StormByte/visibility.h" "")
write_fixture_library("${_prefix}/lib/${_library_prefix}StormByte${_suffix}" "${_shared}")

	file(WRITE "${_prefix}/bin/StormByte.dll" "")
endif()

foreach(_component Logger System Buffer Crypto Network FixtureBroken FixtureHeaderOnly FixtureLibraryOnly)
	string(TOLOWER "${_component}" _lower)
	file(MAKE_DIRECTORY "${_prefix}/include/StormByte/${_lower}")
	if(NOT _component STREQUAL "FixtureLibraryOnly")
		file(WRITE "${_prefix}/include/StormByte/${_lower}/visibility.h" "")
	endif()
	if(NOT _component STREQUAL "FixtureHeaderOnly")
		write_fixture_library("${_prefix}/lib/${_library_prefix}StormByte-${_component}${_suffix}" "${_shared}")
		if(WIN32 AND NOT STORMBYTE_TEST_MINGW AND STORMBYTE_TEST_STATIC)
			file(WRITE "${_prefix}/bin/StormByte-${_component}.dll" "")
		endif()
	endif()
	file(WRITE "${_modules}/FindStormByte${_component}.cmake"
		"stormbyte_find_component(${_component})\n")
endforeach()
file(WRITE "${_modules}/FindStormByteBuffer.cmake"
	"stormbyte_find_component(Buffer DEPENDS Logger System)\n")
file(WRITE "${_modules}/FindStormByteCrypto.cmake"
	"stormbyte_find_component(Crypto DEPENDS Buffer)\n")
file(WRITE "${_modules}/FindStormByteNetwork.cmake"
	"stormbyte_find_component(Network DEPENDS Buffer Crypto)\n")
file(WRITE "${_modules}/FindStormByteFixtureBroken.cmake"
	"stormbyte_find_component(FixtureBroken DEPENDS FixtureMissing)\n")

file(MAKE_DIRECTORY "${_prefix}/include/StormByte/database")
file(WRITE "${_prefix}/include/StormByte/database/visibility.h" "")
write_fixture_library("${_prefix}/lib/${_library_prefix}StormByte-Database${_suffix}" "${_shared}")

if(WIN32 AND NOT STORMBYTE_TEST_MINGW)
	foreach(_component FixtureStatic FixtureShared FixtureOverrideOn FixtureOverrideOff)
		string(TOLOWER "${_component}" _lower)
		file(MAKE_DIRECTORY "${_prefix}/include/StormByte/${_lower}")
		file(WRITE "${_prefix}/include/StormByte/${_lower}/visibility.h" "")
		if(_component STREQUAL "FixtureOverrideOn")
			write_fixture_library("${_prefix}/lib/StormByte-${_component}.lib" TRUE "RenamedDLL")
		elseif(_component STREQUAL "FixtureShared")
			write_fixture_library("${_prefix}/lib/StormByte-${_component}.lib" TRUE "stormbyte-fixtureshared")
		elseif(_component STREQUAL "FixtureOverrideOff")
			write_fixture_library("${_prefix}/lib/StormByte-${_component}.lib" TRUE)
		else()
			write_fixture_library("${_prefix}/lib/StormByte-${_component}.lib" FALSE "StormByte-FixtureStaticExtra")
			file(WRITE "${_prefix}/bin/StormByte-${_component}.dll" "")
		endif()
		file(WRITE "${_modules}/FindStormByte${_component}.cmake"
			"stormbyte_find_component(${_component})\n")
	endforeach()
endif()

		"${_other_prefix}/share/cmake/Modules/FindStormByteLogger.cmake")

foreach(_target StormByte StormByte-Base StormByte::Base StormByte::Logger StormByte::System StormByte::Buffer StormByte::Crypto StormByte::Network)
	if(NOT TARGET ${_target})
		message(FATAL_ERROR "Missing transitive target ${_target}")
	endif()
endforeach()
foreach(_component Buffer Network)
	get_target_property(_links StormByte-${_component} INTERFACE_LINK_LIBRARIES)
	if(_component STREQUAL "Buffer")
		set(_expected "StormByte;StormByte::Logger;StormByte::System")
	else()
		set(_expected "StormByte;StormByte::Buffer;StormByte::Crypto")
	endif()
	if(NOT _links STREQUAL _expected)
		message(FATAL_ERROR "Incorrect ${_component} dependencies: ${_links}")
	endif()
endforeach()
get_target_property(_includes StormByte INTERFACE_INCLUDE_DIRECTORIES)
if(NOT _includes STREQUAL "${_prefix}/include")
	message(FATAL_ERROR "Core include directory must contain StormByte/: ${_includes}")
endif()
if(DEFINED STORMBYTE_TEST_CORE_OVERRIDE)
	check_visibility(StormByte STORMBYTE_SHARED "${STORMBYTE_TEST_CORE_OVERRIDE}")
else()
	check_visibility(StormByte STORMBYTE_SHARED "${_shared}")
endif()

	string(TOUPPER "${_component}" _upper)
	check_visibility(StormByte-${_component} "STORMBYTE_${_upper}_SHARED" "${_shared}")
endforeach()

	check_visibility(StormByte-FixtureOverrideOn STORMBYTE_FIXTUREOVERRIDEON_SHARED TRUE)
	check_visibility(StormByte-FixtureOverrideOff STORMBYTE_FIXTUREOVERRIDEOFF_SHARED FALSE)

		message(FATAL_ERROR "Missing or incomplete component ${_missing} was reported as available")
	endif()
endforeach()
